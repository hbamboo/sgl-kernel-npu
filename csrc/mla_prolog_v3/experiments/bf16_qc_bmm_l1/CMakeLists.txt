cmake_minimum_required(VERSION 3.20)
project(bf16_qc_bmm_l1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_BUILD_TYPE RELEASE)
set(SOC_VERSION Ascend950PR_9599)
set(ASCEND_HOME_PATH /usr/local/Ascend/cann-9.1.0)
set(INC ${ASCEND_HOME_PATH}/aarch64-linux/include)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
include(${ROOT}/cmake/config_envs.cmake)
include(${ROOT}/cmake/config_ascend.cmake)
ascendc_library(qc_bmm_kernel STATIC kernel.cpp)
ascendc_include_directories(qc_bmm_kernel PRIVATE ${ROOT}/csrc/mla_prolog_v3/op_kernel ${ROOT}/csrc/mla_prolog/op_kernel ${ROOT}/csrc/mla_prolog/op_kernel/arch35 ${ROOT}/csrc/utils/kernel ${INC})
set(QC_N 192 CACHE STRING "First GEMM N tile")
set(QC_K 64 CACHE STRING "First GEMM K tile")
set(QC_K_OUTER 0 CACHE STRING "Reuse A in L0 over N before advancing K")
ascendc_compile_definitions(qc_bmm_kernel PRIVATE -DEXP_QC_K_OUTER=${QC_K_OUTER} -DHAVE_WORKSPACE -DHAVE_TILING -DEXP_QC_N=${QC_N} -DEXP_QC_K=${QC_K})
ascendc_compile_options(qc_bmm_kernel PRIVATE --cce-auto-sync=off)
add_library(fused_qc_bmm SHARED host.cpp)
target_include_directories(fused_qc_bmm PRIVATE ${ROOT}/include ${ROOT}/csrc/utils ${TORCH_DIR}/include ${TORCH_DIR}/include/torch/csrc/api/include ${TORCH_NPU_DIR}/include ${INC} ${INC}/external ${INC}/experiment/platform ${INC}/experiment/runtime)
target_link_directories(fused_qc_bmm PRIVATE ${TORCH_DIR}/lib ${TORCH_NPU_DIR}/lib ${ASCEND_HOME_PATH}/aarch64-linux/lib64)
target_link_libraries(fused_qc_bmm PRIVATE qc_bmm_kernel torch torch_cpu c10 torch_npu ascendcl tiling_api register platform ascendalog dl)
